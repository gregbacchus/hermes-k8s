{{- if .Values.gateway.enabled }}
1. Get the gateway API server URL:
{{- if contains "NodePort" .Values.gateway.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "hermes.fullname" . }}-gateway)
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.gateway.service.type }}
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "hermes.fullname" . }}-gateway -o jsonpath='{.status.loadBalancer.ingress[0].ip}')
  echo http://$SERVICE_IP:{{ .Values.gateway.service.port }}
{{- else }}
  kubectl --namespace {{ .Release.Namespace }} port-forward svc/{{ include "hermes.fullname" . }}-gateway {{ .Values.gateway.service.port }}:{{ .Values.gateway.service.port }}
{{- end }}
{{- end }}

{{- if and .Values.gateway.enabled .Values.gateway.dashboard.enabled }}

{{- if contains "NodePort" .Values.gateway.dashboard.service.type }}

{{- if .Values.gateway.apiServer.enabled }}
{{- if not (or .Values.gateway.apiServer.key .Values.gateway.apiServer.keySecretRef.name .Values.secret.enabled) }}
Note: the gateway API server requires an API_SERVER_KEY of at least 16 characters.
Set gateway.apiServer.key, gateway.apiServer.keySecretRef, or
secret.stringData.API_SERVER_KEY to start it. Shorter keys are silently rejected by
the Hermes image and the pod crash-loops.
{{- end }}
{{- else }}
Note: the API server is disabled (gateway.apiServer.enabled=false), so the gateway's
liveness/readiness probes and helm test hook are not rendered. The Hermes image still
starts the API server if any API_SERVER_KEY reaches the container (e.g. via
secret.stringData or gateway.envFrom) — it ignores API_SERVER_ENABLED=false.
{{- end }}
